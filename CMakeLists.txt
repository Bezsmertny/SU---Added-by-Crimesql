cmake_minimum_required(VERSION 3.20)
project(CrimeSQLRecovery VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(recovery_core src/recovery.cpp src/features.cpp src/mini_games.cpp src/recovery_actions.cpp)
target_include_directories(recovery_core PUBLIC src)
if(MSVC)
  target_compile_options(recovery_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(recovery_core PUBLIC -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  enable_language(RC)
  find_package(Threads REQUIRED)
  add_library(windows_backend src/windows_system.cpp src/windows_features.cpp src/file_watcher.cpp)
  target_compile_definitions(windows_backend PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(windows_backend PUBLIC recovery_core advapi32 shell32 ole32 uuid bcrypt user32 dwmapi Threads::Threads)
  add_executable(CrimeSQLRecovery WIN32 src/main.cpp resources/app.rc)
  target_link_libraries(CrimeSQLRecovery PRIVATE windows_backend gdi32 gdiplus comctl32 comdlg32)
  add_executable(CrimeSQLObserver WIN32 src/observer.cpp resources/app.rc)
  target_link_libraries(CrimeSQLObserver PRIVATE windows_backend comctl32 comdlg32)
  foreach(app CrimeSQLRecovery CrimeSQLObserver)
    target_include_directories(${app} PRIVATE resources)
    if(MINGW)
      target_compile_definitions(${app} PRIVATE NHL_EMBED_MANIFEST)
      target_link_options(${app} PRIVATE -municode -static -static-libgcc -static-libstdc++)
    elseif(MSVC)
      target_sources(${app} PRIVATE resources/app.manifest)
    endif()
  endforeach()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(recovery_tests tests/recovery_tests.cpp)
  target_link_libraries(recovery_tests PRIVATE recovery_core)
  add_test(NAME recovery_tests COMMAND recovery_tests)
  add_executable(feature_tests tests/feature_tests.cpp)
  target_link_libraries(feature_tests PRIVATE recovery_core)
  add_test(NAME feature_tests COMMAND feature_tests)
  add_executable(game_tests tests/game_tests.cpp)
  target_link_libraries(game_tests PRIVATE recovery_core)
  add_test(NAME game_tests COMMAND game_tests)
  if(WIN32)
    add_executable(windows_integration_tests tests/windows_integration_tests.cpp)
    target_link_libraries(windows_integration_tests PRIVATE windows_backend)
    if(MINGW)
      target_link_options(windows_integration_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME windows_integration_tests COMMAND windows_integration_tests)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_compile_definitions(ui_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(ui_tests PRIVATE user32 gdi32)
    if(MINGW)
      target_link_options(ui_tests PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME ui_tests COMMAND ui_tests $<TARGET_FILE:CrimeSQLRecovery>)
    set_tests_properties(ui_tests PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
  endif()
endif()
